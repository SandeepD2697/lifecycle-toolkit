#!/usr/bin/env python3 import sys import os import json import html import yaml import requests def download_from_jfrog(jfrog_url, download_path, jfrog_user, jfrog_api_key): print(f"⬇️ Downloading YAML from JFrog: {jfrog_url}") response = requests.get(jfrog_url, auth=(jfrog_user, jfrog_api_key)) if response.status_code != 200: print(f"❌ Failed to download YAML. Status: {response.status_code}, Body: {response.text}") sys.exit(1) with open(download_path, "wb") as f: f.write(response.content) print(f"✅ YAML downloaded to {download_path}") def generate_html_report(yaml_file, output_file): with open(yaml_file) as f: data = yaml.safe_load(f) status = data.get("status", {}) timeframe = status.get("timeframe", {}) raw_json = status.get("raw", "{}") state = status.get("state", "") pass_status = status.get("pass", False) warning_status = status.get("warning", False) # Parse the raw field raw_data = {} try: raw_data = json.loads(raw_json) except Exception as e: print("⚠️ Failed to parse raw field:", e) objective_results = raw_data.get("objectiveResults", []) analysis_name = data["metadata"]["name"] namespace = data["metadata"]["namespace"] html_content = f""" Keptn Analysis Report - {analysis_name}

Keptn Analysis Report

Analysis: {analysis_name}

Namespace: {namespace}

From: {timeframe.get("from", "")}

To: {timeframe.get("to", "")}

State: {state}

Total Score: {raw_data.get("totalScore", "N/A")} / {raw_data.get("maximumScore", "N/A")}

Passed: {pass_status} | Warning: {warning_status}

""" for obj in objective_results: objective = obj.get("objective", {}).get("analysisValueTemplateRef", {}).get("name", "-") value = obj.get("value", "-") result = obj.get("result", {}) query = html.escape(obj.get("query", "")) pass_result = result.get("pass", False) warn_result = result.get("warning", False) fail_result = result.get("failResult", {}).get("fulfilled", False) # Fixed threshold (if defined) target = obj.get("objective", {}).get("target", {}) fixed_value = "-" if "failure" in target: failure = target.get("failure", {}) if "greaterThan" in failure: fixed_value = failure["greaterThan"].get("fixedValue") elif "lessThan" in failure: fixed_value = failure["lessThan"].get("fixedValue") html_content += f""" """ html_content += "
Objective Value Fixed Target Pass Warning Failure Query
{objective} {value} {fixed_value} {pass_result} {warn_result} {fail_result} 
{query}
" with open(output_file, "w") as f: f.write(html_content) print(f"✅ HTML report generated: {output_file}") def upload_to_jfrog(upload_url, local_file, jfrog_user, jfrog_api_key): print(f"⬆️ Uploading report to JFrog: {upload_url}") with open(local_file, 'rb') as f: response = requests.put(upload_url, auth=(jfrog_user, jfrog_api_key), data=f) if response.status_code in [200, 201]: print(f"✅ Uploaded successfully: {upload_url}") else: print(f"❌ Upload failed. Status: {response.status_code}, Response: {response.text}") sys.exit(1) # --- MAIN --- if len(sys.argv) != 6: print("Usage: generate-keptn-html-report.py ") sys.exit(1) download_url = sys.argv[1] yaml_file = sys.argv[2] html_file = sys.argv[3] upload_url = sys.argv[4] jfrog_user = sys.argv[5] jfrog_api_key = os.getenv("JFROG_API_KEY") if not jfrog_api_key: print("❌ Missing environment variable: JFROG_API_KEY") sys.exit(1) download_from_jfrog(download_url, yaml_file, jfrog_user, jfrog_api_key) generate_html_report(yaml_file, html_file) upload_to_jfrog(upload_url, html_file, jfrog_user, jfrog_api_key)
